# Icons Gallery (`@bobslab/icons`)

Accessible, scalable SVG icons designed for modern user interfaces.

## Usage

```tsx
import {
  Logo,
  SpinnerIcon,
  CheckIcon,
  CopyIcon,
  CloseIcon,
  SearchIcon,
  ChevronDownIcon,
  ChevronUpIcon,
  ChevronLeftIcon,
  ChevronRightIcon,
  ChevronsUpDownIcon,
} from "@bobslab/icons";

export function IconsDemo() {
  return (
    <div className="flex flex-wrap gap-4 items-center">
      <Logo size={28} />
      <SpinnerIcon size={20} />
      <CheckIcon size={20} />
      <CopyIcon size={20} />
      <CloseIcon size={20} />
      <SearchIcon size={20} />
      <ChevronDownIcon size={20} />
    </div>
  );
}
```
