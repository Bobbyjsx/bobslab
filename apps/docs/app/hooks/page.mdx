# React Hooks (`@bobslab/hooks`)

SSR-safe, production-tested hooks for React 19 and 18.

## `useDebounce`

```tsx
import { useDebounce } from "@bobslab/hooks";

const debouncedSearch = useDebounce(searchTerm, 300);
```

## `useLocalStorage`

```tsx
import { useLocalStorage } from "@bobslab/hooks";

const [theme, setTheme] = useLocalStorage("app-theme", "system");
```

## `useClipboard`

```tsx
import { useClipboard } from "@bobslab/hooks";

const { copied, copy } = useClipboard({ timeout: 2000 });
<button onClick={() => copy("Text to copy")}>
  {copied ? "Copied!" : "Copy"}
</button>
```

## `useCountdown`

```tsx
import { useCountdown } from "@bobslab/hooks";

const { countdown, isRunning, start, reset } = useCountdown({
  seconds: 60,
  autoStart: true,
  onComplete: () => console.log("Timer complete!"),
});
```

## `useIsMobile`

```tsx
import { useIsMobile } from "@bobslab/hooks";

const isMobile = useIsMobile(768);
```
